cmake_minimum_required(VERSION 3.28)
project(CaptionProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ARIBCC_NO_RENDERER ON CACHE BOOL "" FORCE)
set(ARIBCC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ARIBCAPTION_SOURCE}" aribcaption)
add_executable(CaptionProbe CaptionProbe.cpp)
target_link_libraries(CaptionProbe PRIVATE aribcaption::aribcaption)
target_include_directories(CaptionProbe PRIVATE "${ARIBCAPTION_SOURCE}/test/sample_data/include")
target_compile_definitions(CaptionProbe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
